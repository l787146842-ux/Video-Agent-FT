> **判定：`读了判据 → 推理对了 → 写出过完全合规的稿 → 该稿被机械续跑丢弃 → 从零重拟时越界`（第四档，非起稿版的第三档）。**

```python
# 规格结构度量（2026-09-22 批R2，事故 9999/Q2）
# 依据：GOVERNANCE §二「能代码校验就下沉」+ «确定性三问»（标题数/条目数
# = 可算+可判+零创作空间 → 收归系统）。数值来源 = flova N=2 样本观测区间
# 上界（标题 1–2 / 条目 10–11），非条款（见 T11 §6.1 K′15）。
_SPEC_H2_MAX = 1          # 二级标题上限（对齐 06 样本的 1 个 `## 分集结构`）
_SPEC_ENTRY_MAX = 12      # 维度条目上限（区间上界 11 + 1 容差）


def spec_structure_metrics(text: str) -> Dict[str, int]:
    """规格文档结构度量（纯函数、无副作用、可单测）。

    返回 {h1, h2, entries, over}：
      h1/h2  = 一级/二级标题数
      entries= 无序条目数（`- ` 行首）
      over   = 超限项数（0 = 全部在限内）
    """
    s = str(text or "")
    h1 = len(re.findall(r"(?m)^# ", s))
    h2 = len(re.findall(r"(?m)^## ", s))
    entries = len(re.findall(r"(?m)^- ", s))
    over = (1 if h2 > _SPEC_H2_MAX else 0) + (1 if entries > _SPEC_ENTRY_MAX else 0)
    return {"h1": h1, "h2": h2, "entries": entries, "over": over}
```
